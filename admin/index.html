oakviewfreight.com Lead Capture Snippet

Paste this JS handler into your oakviewfreight.com enquiry form so website leads automatically appear in Tab 2 (using text/plain to avoid CORS preflight blocks). 

<script>
async function submitOakviewLead(formData) {
  const payload = {
    token: "YOUR_SECRET_TOKEN",
    action: "addLead",
    source: new URLSearchParams(location.search).get("utm_source") || "Website",
    company: formData.company,
    contactName: formData.name,
    email: formData.email,
    phone: formData.whatsapp,
    service: formData.service, // e.g. "UK Amazon FBA IOR/DDP"
    notes: formData.message
  };
  await fetch("YOUR_GOOGLE_APPS_SCRIPT_WEB_APP_URL", {
    method: "POST",
    headers: { "Content-Type": "text/plain;charset=utf-8" },
    body: JSON.stringify(payload)
  });
}
</script>
